cmake_minimum_required(VERSION 3.22)
project(openscreen_input)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(KDE_INSTALL_PLUGINDIR "lib/qt6/plugins" CACHE PATH "Qt plugin installation directory")
find_package(Qt6 REQUIRED COMPONENTS Core Network Widgets)
find_package(KWin 6.8 REQUIRED)
find_package(KF6CoreAddons REQUIRED)
kcoreaddons_add_plugin(openscreen_input SOURCES openscreen_input.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
set_target_properties(openscreen_input PROPERTIES PREFIX "")
target_link_libraries(openscreen_input KWin::kwin Qt6::Network)
